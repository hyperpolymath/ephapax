module p1

data Option(a) = None | Some(a)

fn unwrap_or(opt: Option(I32), d: I32): I32 =
  match opt of
  | None => d
  | Some(v) => v
  end

fn fact(n: I32): I32 =
  if n <= 1 then 1 else n * fact(n - 1)

fn main(): I32 =
  let! x = 40
  let y = unwrap_or(Some(2), 0)
  let z = fact(5)
  x + y + z
